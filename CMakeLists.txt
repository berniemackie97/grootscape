cmake_minimum_required(VERSION 3.28)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" GROOTSCAPE_VERSION LIMIT_COUNT 1)
if(CMAKE_HOST_APPLE AND NOT CMAKE_TOOLCHAIN_FILE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum native macOS version" FORCE)
endif()
project(Grootscape VERSION ${GROOTSCAPE_VERSION} LANGUAGES C CXX)
include(CTest)
include(cmake/Quality.cmake)
option(GROOTSCAPE_RENDERER "Build SDL renderer and desktop host" ON)
option(GROOTSCAPE_TIDY "Run clang-tidy on first-party C++" OFF)
set(GROOTSCAPE_SANITIZER "" CACHE STRING "Sanitizer: address,undefined or thread")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
configure_file(src/client/include/grootscape/version.h.in generated/grootscape/version.h @ONLY)
add_library(grootscape_protocol STATIC src/client/cpp/protocol.cpp)
target_include_directories(grootscape_protocol PUBLIC src/client/include)
grootscape_quality(grootscape_protocol)
if(ANDROID)
 add_library(grootscape_native SHARED src/client/cpp/native.cpp)
else()
 add_library(grootscape_native STATIC src/client/cpp/native.cpp)
endif()
target_include_directories(grootscape_native PUBLIC src/client/include "${CMAKE_CURRENT_BINARY_DIR}/generated")
grootscape_quality(grootscape_native)
if(GROOTSCAPE_RENDERER)
 include(FetchContent)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/archive/d9d5536704d585616d4db3c8ba3c4ff6fc2757e1.tar.gz URL_HASH SHA256=44b617821ea70cb6515a05f962d4133a5e32c1011c98a2f1daea01a187279a43)
 FetchContent_MakeAvailable(SDL3)
 add_library(grootscape_engine STATIC src/client/cpp/renderer.cpp)
 target_include_directories(grootscape_engine PUBLIC src/client/include "${CMAKE_CURRENT_BINARY_DIR}/generated")
 target_link_libraries(grootscape_engine PUBLIC SDL3::SDL3)
 grootscape_quality(grootscape_engine)
 if(EMSCRIPTEN)
  add_executable(grootscape_web src/client/cpp/web.cpp)
  target_link_libraries(grootscape_web PRIVATE grootscape_engine)
  grootscape_quality(grootscape_web)
  target_compile_options(grootscape_web PRIVATE -fexceptions)
  target_compile_options(grootscape_engine PRIVATE -fexceptions)
  target_link_options(grootscape_web PRIVATE -fexceptions -sMODULARIZE=1 -sEXPORT_ES6=1 -sALLOW_MEMORY_GROWTH=1 -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2 -sEXPORTED_RUNTIME_METHODS=cwrap)
  set_target_properties(grootscape_web PROPERTIES SUFFIX .js RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/apps/web/pkg")
 else()
  add_executable(grootscape-client src/client/cpp/desktop.cpp)
  target_link_libraries(grootscape-client PRIVATE grootscape_engine)
  grootscape_quality(grootscape-client)
 endif()
endif()
if(BUILD_TESTING AND NOT ANDROID AND NOT EMSCRIPTEN)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate-fixture-tests.py" "${CMAKE_CURRENT_SOURCE_DIR}/protocol/fixtures/v1.json" "${CMAKE_CURRENT_BINARY_DIR}/generated/fixtures.cpp" COMMAND_ERROR_IS_FATAL ANY)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS protocol/fixtures/v1.json tools/generate-fixture-tests.py)
 add_executable(client-tests tests/client/core.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/fixtures.cpp")
 target_link_libraries(client-tests PRIVATE grootscape_protocol grootscape_native)
 grootscape_quality(client-tests)
 add_executable(protocol-peer tests/client/protocol-peer.cpp)
 target_link_libraries(protocol-peer PRIVATE grootscape_protocol)
 grootscape_quality(protocol-peer)
 add_test(NAME client-contracts COMMAND client-tests)
 if(GROOTSCAPE_RENDERER)
  add_executable(renderer-tests tests/client/renderer.cpp)
  target_link_libraries(renderer-tests PRIVATE grootscape_engine)
  grootscape_quality(renderer-tests)
  add_test(NAME renderer-lifecycle COMMAND renderer-tests)
  set_tests_properties(renderer-lifecycle PROPERTIES ENVIRONMENT "SDL_RENDER_DRIVER=software" TIMEOUT 30)
  add_test(NAME desktop-version COMMAND grootscape-client --version)
  set_tests_properties(desktop-version PROPERTIES PASS_REGULAR_EXPRESSION "grootscape-client ${PROJECT_VERSION}")
 endif()
endif()
